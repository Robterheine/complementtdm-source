$SIZES      LVR=95 LNP4=7000
$PROBLEM    PK
$INPUT      ID TIME AMT RATE DV MDV EVID CMT WT SEX AGE II SS
            CH50 OCC DISEASE 
$DATA      PNHAHUS_okt22.csv  
$SUBROUTINE ADVAN13 TOL=6
$ABBREVIATED DERIV2=NOCOMMON
$MODEL      
COMP=(CENTRAL)
COMP=(PERIPHERAL)



$PK 


; IOV
IF(OCC.EQ.1)	IOV	=	ETA(1)
IF(OCC.EQ.2)	IOV	=	ETA(2)
IF(OCC.EQ.3)	IOV	=	ETA(3)
IF(OCC.EQ.4)	IOV	=	ETA(4)
IF(OCC.EQ.5)	IOV	=	ETA(5)
IF(OCC.EQ.6)	IOV	=	ETA(6)



PNH = 0
IF (DISEASE.EQ.2) PNH=1 ; IF DISEASE IS 1 DISEASE=AHUS, IF DISEASE IS 2 DISEASE=PNH

IF (PNH.EQ.0) CL = THETA(1) * ((WT/70)**0.75) * EXP(ETA(7)) * EXP(IOV) ; IF PNH=0 THEN DISEASE IS AHUS
IF (PNH.EQ.1) CL = THETA(1) * ((WT/70)**0.75) * EXP(ETA(7))   ; IF PNH=1 THEN DISEASE IS PNH

V1 = THETA(2) * (WT/70) * EXP(ETA(8))	
V2 = THETA(5) * (WT/70)
Q = THETA(6) * (WT/70)**0.75	
VM = THETA(3) * (WT/70)**0.75
KM = THETA(4)

V=V1
S1=V

ICL=CL
IV1=V1
IV2=V2
IQ=Q

KEL=CL/V1	;Elimination rate constant
K12=Q/V1
K21=Q/V2


$ERROR 	
IPRED=F
Y=IPRED+ERR(1)+(IPRED*ERR(2))



$THETA  
(0, 0.1626) ; 1 CL
(0, 4.299) ; 2 V1
(0, 25.64) ; 3 VM
(0, 13.54) ; 4 KM
(0, 2.63) ; 5 V2
(0, 0.6243) ;  Q


$DES 
C1=A(1)/S1				
DADT(1)= -KEL*A(1) - ((VM*C1)/(KM+C1)) -K12*A(1) + K21*A(2)
DADT(2) = -K21*A(2) + K12*A(1)	


$OMEGA  BLOCK(1)
 0.112  ;      1 IOV
$OMEGA  BLOCK(1) SAME ; 2 IOV
$OMEGA  BLOCK(1) SAME ; 3 IOV
$OMEGA  BLOCK(1) SAME ; 4 IOV
$OMEGA  BLOCK(1) SAME ; 5 IOV
$OMEGA  BLOCK(1) SAME ; 6 IOV

$OMEGA 
 0.1546 ;     IIV CL
 0.0812  ;     IIV V1


$SIGMA  
19.1;        ADD
 0.0185  ;       PROP

$ESTIMATION METHOD=1 INTERACTION NOABORT MAXEVAL=5000 NSIG=2 SIGL=6
PRINT=5
$COVARIANCE PRINT=E MATRIX=S UNCONDITIONAL
