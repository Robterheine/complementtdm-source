
$PROBLEM    Ravulizumab reestemation
$INPUT      ID DROP WT DROP TIME AMT RATE DV CMT MDV EVID
$DATA      simset01.csv IGNORE=@
$SUBROUTINE ADVAN3 TRANS4
$PK

ALLOCL=(WT/70)**THETA(1)
ALLOV=(WT/70)**THETA(2)

CL=THETA(3)*ALLOCL*EXP(ETA(1))
V1=THETA(4)*ALLOV*EXP(ETA(2))
V2=THETA(5)*ALLOV
Q=THETA(6)*ALLOCL

S1=V1

D1=2.5 ; 2.5 HR INFUSION
 
$ERROR
IPRED=F
IF (CMT.EQ.1) Y=F+F*ERR(1)


$THETA  
0.833; 1 ALLOCL
0.620; 2 ALLOV
0.00342 ; 3 CL LITER PER HOUR
3.43 ; 4 V1 IN LITER
1.86; 5 V2 IN LITER
0.0162 ; 6 Q IN LITER PER HOUR


$OMEGA  BLOCK(2)
 0.0569  ;       1 CL
 0.0161 0.0264  ;        2 V


$SIGMA  2.39E-02 ; PROP ERR PK

$ESTIMATION METHOD=1 INTERACTION NOHABORT MAXEVAL=3000 PRINT=10
$COVARIANCE MATRIX=S UNCONDITIONAL PRINT=E
$TABLE      ID WT TIME AMT RATE DV CMT MDV EVID IPRED CWRES NOAPPEND
            NOPRINT ONEHEADER FILE=sdtabravuest01.tab
