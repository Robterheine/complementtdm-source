$PROBLEM    PKPD CROVALIMAB
$INPUT      ID AGE SEX WT HEIGHT TIME AMT CMT EVID MDV DV
$DATA      Standard_crovalimab.csv IGNORE=@
$SUBROUTINE ADVAN13 TOL=9
; --- PK
$MODEL      COMP=(SC) 		; 1 SC DEPOT
            COMP=(CENTRAL) 	; 2 CENTRAL DOSING AND OBSERVATION COMPARTMENT
            COMP=(PERI) 	; 3 PERIPHERAL PK
           

$PK
; --- TIME AFTER DOSE CALCULATION
IF(AMT.GT.0) THEN 
 TDOS=TIME 
 TAD=0.0 
 ENDIF 
 IF (AMT.EQ.0) TAD=TIME-TDOS 
 


ALLOCL=(WT/75)**0.75	; IMPACT OF TOTAL BODY WEIGHT ON CLEARANCE
ALLOV=(WT/75)**0.684	; IMPACT OF TOTAL BODY WEIGHT ON VOLUME OF DISTRIBUTION
AGEKA=(-0.459)         	; IMPACT OF AGE ON ABSORPTION RATE

CL=THETA(1)*ALLOCL*EXP(ETA(1))
V2=THETA(2)*ALLOV*EXP(ETA(2))
Q=THETA(3)*ALLOCL*EXP(ETA(3))
V3=THETA(4)*ALLOV*EXP(ETA(4))
KA=THETA(5)*((AGE/36)**AGEKA)*EXP(ETA(5))

LOGITF1=LOG(THETA(6)/(1-THETA(6)))+ETA(6)	; LOGIT TRANSFORMATION OF BIOAVAILABILITY TO CONSTRAIN <1
F1=1/(1+EXP(-LOGITF1)) ; BIOAVAILABILITY OF CROVALIMAB IN COMPARTMENT 1 (SUBCUTANEOUS)



S2=V2

;--- MASS TRANSPORT
K12=KA
K23=Q/V2
K32=Q/V3
K20=CL/V2


$DES
; --- PK
DADT(1)=-K12*A(1)
DADT(2)=-K23*A(2)-K20*A(2)+K12*A(1)+K32*A(3)
DADT(3)=-K32*A(3)+K23*A(2)



$ERROR
CROVAL=A(2)/V2
IPRED=CROVAL
IF (CMT.EQ.2) Y=CROVAL+(CROVAL*ERR(1))+(ERR(2)*EXP(ETA(7)))


$THETA  
 0.0791 ; 1 CL LITER/DAY
 3.23 ; 2 V2 LITER
 0.168 ; 3 Q LITER/DAY
 2.32 ; 4 V3 LITER
 0.126 ; 5 KA 1/DAY
 0.83 ; 6 F1 FRACTION


$OMEGA  
 0.0424  	; 1 CL
 0.0502  	; 2 V2
 0.341  	; 3 Q
 0.498  	; 4 V3
 0.147  	; 5 KA
 1.35  		; 6 F1
 0.619  	; 7 IIV ADD ERR
 
$SIGMA  0.01277  ; PROP ERR PK
 1.02  ; ADD ERR PK MG/L
$SIMULATION ONLYSIM SUBPROBLEMS=1 (38729) (4598)
$TABLE ID AGE SEX WT HEIGHT TIME AMT CMT EVID MDV DV IPRED CROVAL LDH I EFF F1 ADA NOPRINT ONEHEADER FILE=sdtabrun09.tab
